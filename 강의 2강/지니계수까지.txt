#전체 지니 불순도
ice_count = sum(y_train == 0) # 훈련 데이터 아이스 개수
hot_count = sum(y_train == 1) # 훈련 데이터 핫 개수
total = len(y_train)
p_ice = ice_count / total
p_hot = hot_count / total
gini_parent = 1 - (p_ice**2 + p_hot**2)
print(f"Gini(parent) = {gini_parent:.4f}")

#기온 >= 22
threshold = 22
left = y_train[X_train["temp"] < threshold] # 기온 < 22
right = y_train[X_train["temp"] >= threshold] # 기온 >= 22
# 왼쪽 자식 지니
p_ice_left = sum(left == 0) / len(left)
p_hot_left = sum(left == 1) / len(left)
gini_left = 1 - (p_ice_left**2 + p_hot_left**2)
# 오른쪽 자식 지니
p_ice_right = sum(right == 0) / len(right)
p_hot_right = sum(right == 1) / len(right)
gini_right = 1 - (p_ice_right**2 + p_hot_right**2)
# 가중 평균 지니
gini_children = (len(left)/total)*gini_left + (len(right)/total)*gini_right
# 정보 이득
info_gain = gini_parent - gini_children
print(f"Info Gain (temp >= 22°C) = {info_gain:.4f}")
